---
title: Build an Extension
description: Step-by-step guide to authoring reusable Effect-native extensions.
---

This guide walks through creating a reusable extension that packages a database table, an Effect Schema contract, and an action factory.

<Steps>

### Step 1: Declare the Database Table

Create `schema.ts` with your Drizzle table definition:

```ts title="src/extensions/notes/schema.ts"
import { sqliteTable, text } from "drizzle-orm/sqlite-core";

export const notesTable = sqliteTable("notes", {
  id: text("id").primaryKey(),
  targetId: text("target_id").notNull(),
  content: text("content").notNull(),
  createdAt: text("created_at").notNull(),
});
```

### Step 2: Define the Browser-Safe Contract

Create `contract.ts` using `crudContract` and declare custom `Schema.TaggedError` classes. Keep this module free of database drivers so it can be imported safely by frontend applications:

```ts title="src/extensions/notes/contract.ts"
import { crudContract } from "@brickkit/crud";
import { Schema } from "effect";
import { notesTable } from "./schema";

export class InvalidNote extends Schema.TaggedError<InvalidNote>()(
  "InvalidNote",
  { message: Schema.String },
  { httpApiStatus: 400 },
) {}

export const NotesContract = crudContract({
  name: "note",
  path: "/api/notes",
  table: notesTable,
  id: "id",
  errors: [InvalidNote],
  fields: {
    id: { read: true, filter: ["eq"] },
    targetId: { read: true, create: true, filter: ["eq"] },
    content: { read: true, create: true, update: true },
    createdAt: { read: true, sort: true },
  },
});
```

### Step 3: Implement the Handler Factory

In `index.ts`, export a factory function that accepts configuration (or uses host database tags) and returns implemented actions:

```ts title="src/extensions/notes/index.ts"
import { sqliteStore } from "@brickkit/crud";
import { Effect } from "effect";
import { notesTable } from "./schema";
import { NotesContract, InvalidNote } from "./contract";
import { AppDatabase } from "../../db";

export { NotesContract, InvalidNote } from "./contract";
export { notesTable } from "./schema";

export function notesExtension() {
  return NotesContract.implement({
    store: sqliteStore(notesTable, AppDatabase),
    access: "public",
    hooks: {
      beforeCreate: ({ data }) =>
        Effect.gen(function* () {
          const content = String(data.content ?? "").trim();
          if (!content) {
            return yield* Effect.fail(
              new InvalidNote({ message: "Note content cannot be empty" }),
            );
          }
          return {
            ...data,
            id: crypto.randomUUID(),
            content,
            createdAt: new Date().toISOString(),
          };
        }),
    },
  });
}
```

### Step 4: Mount on a Service

In your host application, invoke the factory and register its actions onto a service:

```ts title="src/app.ts"
import { Brick } from "@brickkit/core";
import { notesExtension } from "./extensions/notes";
import { DatabaseLive } from "./db";

const noteActions = notesExtension();

export const notesService = Brick.service("notes", {
  actions: noteActions,
});

export default Brick.app({
  services: [notesService],
  layer: DatabaseLive,
});
```

</Steps>

## Testing Your Extension

Test extensions locally using Bun and Effect test runners:

```bash
bun test src/extensions/notes
```

You can execute actions directly in-process by supplying `DatabaseLive` with `Effect.provide`, testing both valid flows and tagged error conditions.
