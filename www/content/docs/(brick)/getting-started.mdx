---
title: Getting started
description: Install core, CLI, and the optional CRUD plugin
---

```sh
bun add @brickkit/core @brickkit/cli @brickkit/crud
```

```ts
import {
  defineDatabase,
  defineService,
  sqliteTable,
  text,
} from "@brickkit/core";
import { crud } from "@brickkit/crud";

const pages = sqliteTable("pages", {
  id: text("id")
    .primaryKey()
    .$defaultFn(() => crypto.randomUUID()),
  title: text("title").notNull(),
  email: text("email"),
  internalNotes: text("internal_notes"),
});
const database = defineDatabase({ tables: { pages } });
const service = defineService("pages", { database });
const page = service.resource({ name: "page", table: pages, id: pages.id }).use(
  crud({
    fields: {
      id: { read: true, filter: ["eq", "in"] },
      title: {
        read: true,
        create: true,
        update: true,
        filter: ["eq", "contains"],
        sort: true,
      },
      email: {
        read: true,
        create: true,
        update: true,
        scrub: ({ value }) =>
          value === null ? null : value.replace(/^(.).*(@.*)$/, "$1***$2"),
      },
    },
  }),
);
```

Apply your application's schema with your migration tooling before requests.
For disposable local SQLite examples, explicitly call `syncSchema(database.tables,
database.getDb())`; do not run schema synchronization automatically in production.

```ts
import { brick } from "@brickkit/cli";
export default brick({ services: [service], prefix: "/api" });
```

Local actions use `{input,ctx?}`. Browser client calls pass the input directly.
Generate the browser contract with the CLI build instead of importing server
implementation modules into frontend bundles.
