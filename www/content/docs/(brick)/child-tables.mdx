Please see the canonical guide: **[Relational Data Modeling & Joins](/docs/relations)**.

### Schema

The related-table example uses these application-owned tables. `target_id` is a logical reference; the example does not declare a SQL foreign key.

export const articlesSchemaFields = [
  {
    name: "id",
    type: "text",
    description: "Row identifier.",
    isPrimaryKey: true,
    defaultValue: "crypto.randomUUID() (application)",
  },
  {
    name: "title",
    type: "text",
    description: "Display title.",
  },
];

<DatabaseTable name="articles" fields={articlesSchemaFields} />

export const commentsSchemaFields = [
  {
    name: "id",
    type: "text",
    description: "Row identifier.",
    isPrimaryKey: true,
    defaultValue: "crypto.randomUUID() (application)",
  },
  {
    name: "target_id",
    type: "text",
    description:
      "Logical parent article identifier; no SQL foreign-key constraint.",
  },
  {
    name: "body",
    type: "text",
    description: "Text content.",
  },
];

<DatabaseTable name="comments" fields={commentsSchemaFields} />
